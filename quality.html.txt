  OPEN-EQMS — Management Game ERP  ·  Milestone 1 (vertical slice)
  Single self-contained file. No external dependencies, no build step, runs from file://, offline.
  Two-file deployment: open_eqms.html (this) + open_eqms.container (data bundle).

  Architecture (all under the OEQMS namespace, one <script> block below):
    OEQMS.util       - ids, hashing, formatting, escaping
    OEQMS.store      - in-memory world state (objects + relations) and transaction log
    OEQMS.tx         - transaction records for every meaningful change
    OEQMS.health     - deterministic, explainable health / KPI computation (§23 traceable scores)
    OEQMS.container  - .container (de)serialization  [M1: JSON format; ZIP w/ native files = later phase]
    OEQMS.fileio     - open/save .container (File System Access API + download/input fallback)
    OEQMS.ai         - AI context builder + provider adapter (mock provider for M1)
    OEQMS.perm       - permission gate (real permissions must never be bypassed by the game UI, §22)
    OEQMS.ui         - command center rendering + event wiring
    OEQMS.app        - bootstrap & app state

  Rule kept throughout: game PRESENTATION over REAL data + DETERMINISTIC rules + TRACEABLE transactions.

/* ---------------------------------------------------------------- util ---- */
const util = {
  now(){ return new Date().toISOString(); },
  uid(prefix){ return (prefix||'ID') + '-' + Math.random().toString(36).slice(2,8).toUpperCase(); },
  // Small deterministic 32-bit hash for content-addressed attachment ids (§7 provenance).
  hash(str){ let h=0x811c9dc5; for(let i=0;i<str.length;i++){ h^=str.charCodeAt(i); h=Math.imul(h,0x01000193);} return ('0000000'+(h>>>0).toString(16)).slice(-8); },
  esc(s){ return String(s==null?'':s).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); },
  fmtDate(iso){ if(!iso) return '—'; const d=new Date(iso); if(isNaN(d)) return String(iso);
    return d.toLocaleDateString(undefined,{day:'2-digit',month:'short'}) + ' ' + d.toLocaleTimeString(undefined,{hour:'2-digit',minute:'2-digit'}); },
  fmtDay(iso){ if(!iso) return '—'; const d=new Date(iso); if(isNaN(d)) return String(iso);

  titleCase(s){ return String(s||'').replace(/_/g,' ').replace(/\b\w/g,c=>c.toUpperCase()); },

/* --------------------------------------------------------------- store ---- */
/* Universal object store. Every entity is an object; applications are views over it. */
const store = {
  objects:new Map(),
  transactions:[],
  listeners:new Set(),

  load(objectsArray, transactions){
    this.objects.clear();
    (objectsArray||[]).forEach(o=>{
      // normalise shape
      o.properties = o.properties||{}; o.relations = o.relations||[];
      o.permissions = o.permissions||{}; o.history = o.history||[];
      this.objects.set(o.id, o);
    });
    this.transactions = transactions ? transactions.slice() : [];
    this.emit();
  },
  get(id){ return this.objects.get(id) || null; },
  all(){ return [...this.objects.values()]; },
  byType(...types){ const set=new Set(types); return this.all().filter(o=>set.has(o.type)); },

  // outbound + inbound relations, resolved to objects
  related(id){
    const o=this.get(id); if(!o) return {out:[],in:[]};
    const out=(o.relations||[]).map(r=>({type:r.type,obj:this.get(r.to),to:r.to})).filter(r=>r.obj);
    const inn=[];
    for(const other of this.all()){
      if(other.id===id) continue;
      for(const r of (other.relations||[])) if(r.to===id) inn.push({type:r.type,obj:other,from:other.id});
    }
    return {out, in:inn};
  },

  // Controlled mutation — every meaningful change produces a transaction (§16 / North Star §6).
  update(id, field, newValue, user){
    const o=this.get(id); if(!o) return null;
    const path=field.split('.');
    let target=o, key=path[0];
    if(path.length>1){ target=o.properties; key=path[1]; }
    const oldValue=target[key];
    target[key]=newValue;
    const t=tx.create(id,'FIELD_UPDATE',field,oldValue,newValue,user);
    o.history.push({tx:t.transactionId,at:t.timestamp,field,from:oldValue,to:newValue});
    this.emit();
    return t;
  },
  add(obj, user){
    obj.properties=obj.properties||{}; obj.relations=obj.relations||[];
    obj.permissions=obj.permissions||{}; obj.history=obj.history||[];
    this.objects.set(obj.id,obj);
    tx.create(obj.id,'OBJECT_CREATE',null,null,obj.type,user);
    this.emit();
    return obj;
  },
  on(fn){ this.listeners.add(fn); },
  emit(){ this.listeners.forEach(fn=>{ try{fn();}catch(e){console.error(e);} }); },
};

  create(objectId,operation,field,oldValue,newValue,user){

      newValue: newValue===undefined?null:newValue,
      user: user||'system',
      timestamp: util.now(),
    };

    return t;

/* -------------------------------------------------------------- health ---- */
/* Deterministic, explainable scores (North Star §23: never arbitrary game scores). */
const health = {
  machine(m){
    let score=100; const breakdown=[];
    if(m.state==='stopped'){ score-=35; breakdown.push(['Machine stopped', -35]); }
    if(m.state==='warning'){ score-=20; breakdown.push(['State: warning', -20]); }
    const tl=m.properties.toolLifePct;
    if(typeof tl==='number' && tl>=90){ score-=18; breakdown.push(['Tool lifetime '+tl+'%', -18]); }
    else if(typeof tl==='number' && tl>=75){ score-=8; breakdown.push(['Tool lifetime '+tl+'%', -8]); }
    const rel=store.related(m.id).in;
    const openNcr=rel.filter(r=>r.obj.type==='quality_event' && r.obj.state!=='closed').length;
    if(openNcr){ score-=openNcr*12; breakdown.push([openNcr+' open quality event(s)', -openNcr*12]); }
    const oot=store.byType('measurement').filter(x=>x.properties.machine===m.id && x.state==='out_of_tolerance').length;
    if(oot){ score-=oot*10; breakdown.push([oot+' out-of-tolerance measurement(s)', -oot*10]); }
    score=Math.max(0,Math.min(100,Math.round(score)));
    return {score, breakdown};
  },
  supplier(s){
    let score=100; const breakdown=[];
    const ppm=s.properties.ppm, late=s.properties.lateDeliveries;
    if(typeof ppm==='number' && ppm>300){ const p=Math.min(30,Math.round((ppm-300)/20)); score-=p; breakdown.push(['PPM '+ppm+' above target', -p]); }
    if(typeof late==='number' && late>=3){ score-=late*5; breakdown.push([late+' late deliveries', -late*5]); }
    score=Math.max(0,Math.min(100,Math.round(score)));
    return {score, breakdown};
  },
  // KPI roll-up for the dashboard
  kpis(){
    const ncrOpen=store.byType('quality_event').filter(o=>o.state!=='closed').length;
    const tasks=store.byType('task');
    const overdue=tasks.filter(t=>t.properties.overdue).length;
    const machines=store.byType('machine');
    const attn=machines.filter(m=>m.state==='warning'||m.state==='stopped').length;
    const suppliers=store.byType('supplier');
    const avgSup=suppliers.length? Math.round(suppliers.reduce((a,s)=>a+health.supplier(s).score,0)/suppliers.length):0;
    // On-time risk: in-progress POs whose progress rate lags days remaining (simple, explainable)
    const pos=store.byType('production_order').filter(p=>p.state==='in_progress');
    let atRisk=0;
    pos.forEach(p=>{ const pct=(p.properties.done||0)/(p.properties.qty||1);
      const due=new Date(p.properties.due); const daysLeft=(due-Date.now())/86400000;
      if(pct<0.7 && daysLeft<7) atRisk++; });
    return {ncrOpen, overdue, attn, avgSup, atRisk, poActive:pos.length, machines:machines.length};
  }
};

      parts.push({kind:'calc', text:`Computed supplier health ${h.score}% — `+(h.breakdown.map(b=>`${b[0]} (${b[1]})`).join('; ')||'within targets.')});
      parts.push({kind:'rec', text:`Consider a supplier corrective action if PPM stays above target. Decision requires authorised quality approval.`});
      parts.push({kind:'sources', ids:[sel.id]});
    } else if(sel){
      parts.push({kind:'fact', text:`${sel.name} (${util.titleCase(sel.type)}) — state "${util.titleCase(sel.state)}".`});
      const rel=store.related(sel.id);
      parts.push({kind:'infer', text:`It has ${rel.out.length} outbound and ${rel.in.length} inbound relationships in the current world.`});
      parts.push({kind:'sources', ids:[sel.id, ...rel.out.map(r=>r.obj.id).slice(0,4)]});
    } else {
      const k=ctx.kpi;
      parts.push({kind:'fact', text:`Plant overview: ${k.ncrOpen} open quality events, ${k.overdue} overdue task(s), ${k.attn} machine(s) needing attention.`});
      parts.push({kind:'rec', text:`Select a specific object (machine, NCR, supplier) so I can ground an answer in its data and sources.`});
      parts.push({kind:'unknown', text:`Without a selected object I can only report aggregate figures.`});
    }
    return parts;
  }
};

      if(!perm.can('create')) return ui.toast('Your role cannot create tasks.');

    store.byType('quality_event').forEach(o=>evs.push({at:o.properties.opened, id:o.id, label:o.name.split(' — ')[0], color:o.properties.severity==='major'?'var(--bad)':'var(--warn)'}));
    store.byType('maintenance').forEach(o=>evs.push({at:o.properties.start||o.properties.planned, id:o.id, label:o.name.split(' — ')[0], color:'var(--info)'}));
    store.byType('production_order').filter(p=>p.properties.due).forEach(p=>evs.push({at:p.properties.due, id:p.id, label:p.id+' due', color:'var(--accent)'}));
    const valid=evs.filter(e=>e.at && !isNaN(new Date(e.at))).sort((a,b)=>new Date(a.at)-new Date(b.at));
    if(!valid.length){ document.getElementById('tlSub').textContent='no dated events'; return; }
    const t0=new Date(valid[0].at).getTime(), t1=new Date(valid[valid.length-1].at).getTime();
    const span=Math.max(1,t1-t0);
    const width=Math.max(track.clientWidth-28, valid.length*118);
    track.style.minWidth=width+'px';
    document.getElementById('tlSub').textContent=`${valid.length} events · ${util.fmtDay(valid[0].at)} – ${util.fmtDay(valid[valid.length-1].at)}`;
    valid.forEach(e=>{
      const x = 14 + ((new Date(e.at).getTime()-t0)/span) * (width-28);
      const focused = app.state.timelineFocus && (e.id===app.state.timelineFocus || store.related(app.state.timelineFocus).out.some(r=>r.obj.id===e.id) || store.related(app.state.timelineFocus).in.some(r=>r.obj.id===e.id));
      const el=document.createElement('div'); el.className='tl-ev'; el.style.left=x+'px';
      el.style.opacity = (app.state.timelineFocus && !focused) ? '0.35' : '1';
      el.innerHTML=`<span class="pt" style="background:${e.color}"></span><span class="lbl">${util.esc(e.label)}</span><span class="tm">${util.fmtDay(e.at)}</span>`;
      el.onclick=()=>app.select(e.id);
      track.appendChild(el);
    });
  },

/* AI panel controller (kept small, uses ai.* + ui.*) */
const ai_ui = {
  add(html, cls){ const log=document.getElementById('aiLog'); const d=document.createElement('div');
    d.className='ai-msg '+(cls||''); d.innerHTML=html; log.appendChild(d); log.scrollTop=log.scrollHeight; },
  async ask(question){
    if(!question) return;
    this.add(util.esc(question),'u');
    const ctx=ai.buildContext();
    const parts=await ai.provider.respond(question, ctx);
    const kindLabel={fact:'FACT',calc:'CALCULATION',infer:'INFERENCE',rec:'RECOMMENDATION',unknown:'UNKNOWN'};
    const kindCls={fact:'k-fact',calc:'k-calc',infer:'k-infer',rec:'k-rec',unknown:'k-unknown'};
    let html=''; let sources=[];
    parts.forEach(p=>{ if(p.kind==='sources'){ sources=p.ids; return; }
      html+=`<div style="margin-bottom:6px"><span class="ai-kind ${kindCls[p.kind]}">${kindLabel[p.kind]}</span>${util.esc(p.text)}</div>`; });
    if(sources.length){ html+=`<div class="ai-src"><b>Sources:</b> ${sources.map(id=>`<span class="s" data-id="${id}">${util.esc(id)}</span>`).join(', ')}</div>`; }
    this.add(html,'a');
    document.querySelectorAll('#aiLog .ai-src .s[data-id]').forEach(s=>s.onclick=()=>app.select(s.dataset.id));
  }
};

/* ---------------------------------------------------------------- app ---- */
const app = {
  state:{ section:'workspace', selectedId:null, containerName:'demo', containerId:null, createdAt:null,
          dirty:false, attachments:[], timelineFocus:null },
  currentUser(){ return store.get('USER-001') || store.byType('user')[0] || {id:'anon',name:'Guest',properties:{authority:'view'}}; },
  markDirty(){ this.state.dirty=true; ui.renderTop(); },
  markSaved(){ this.state.dirty=false; ui.renderTop(); },

  select(id){ this.state.selectedId=id; this.state.timelineFocus=id;
    ui.renderMain(); ui.renderCtx(); ui.renderTimeline();
    if(id){ const el=document.querySelector(`tr[data-id="${id}"]`); if(el) el.classList.add('sel'); } },

  loadDemo(){
    const raw=document.getElementById('demo-world').textContent;
    const world=JSON.parse(raw);
    store.load(world.objects, []);
    this.state.containerName='demo'; this.state.containerId=util.uid('OEQMS');
    this.state.createdAt=util.now(); this.state.dirty=false; this.state.attachments=[]; this.state.selectedId=null;
  },

  wireChrome(){
    // File menu
    const fileBtn=document.getElementById('fileBtn'), fileMenu=document.getElementById('fileMenu');
    fileBtn.onclick=(e)=>{ e.stopPropagation(); fileMenu.classList.toggle('open'); };
    document.addEventListener('click',()=>fileMenu.classList.remove('open'));
    fileMenu.querySelectorAll('[data-cmd]').forEach(b=>b.onclick=()=>{
      const c=b.dataset.cmd;
      if(c==='new'){ if(this.state.dirty && !confirm('Discard unsaved changes and load the demo world?')) return; this.loadDemo(); ui.renderAll(); ui.toast('Loaded demo world'); }
      else if(c==='open') fileio.open();
      else if(c==='save') fileio.save();
      else if(c==='attach') fileio.attach();
    });
    // Profile
    document.getElementById('profileSel').onchange=(e)=>{
      document.body.className = 'profile-'+e.target.value;
      ui.renderTimeline();
    };
    // Search
    const search=document.getElementById('search'), results=document.getElementById('searchResults');
    const doSearch=()=>{ const q=search.value.trim().toLowerCase(); if(!q){ results.classList.remove('open'); return; }
      const hits=store.all().filter(o=>o.name.toLowerCase().includes(q)||o.id.toLowerCase().includes(q)||o.type.includes(q)).slice(0,12);
      results.innerHTML=hits.map(o=>`<div class="row" data-id="${o.id}">${ui.chip(o.state)}<span>${util.esc(o.name)}</span><small>${util.esc(o.id)}</small></div>`).join('')
        || '<div class="row muted">No matches</div>';
      results.classList.add('open');
      results.querySelectorAll('.row[data-id]').forEach(r=>r.onclick=()=>{ app.select(r.dataset.id); results.classList.remove('open'); search.value=''; });
    };
    search.oninput=doSearch;
    search.onfocus=doSearch;
    document.addEventListener('click',(e)=>{ if(!e.target.closest('.top-search')) results.classList.remove('open'); });
    // AI input
    const aiInput=document.getElementById('aiInput');
    const send=()=>{ const q=aiInput.value.trim(); if(!q) return; aiInput.value=''; ai_ui.ask(q); };
    document.getElementById('aiSend').onclick=send;
    aiInput.onkeydown=(e)=>{ if(e.key==='Enter') send(); };
  },

  init(){
    this.loadDemo();
    this.wireChrome();
    ui.renderAll();
    // open on workspace, nothing selected
  }
};

return { util, store, tx, health, container, fileio, perm, ai, ui, ai_ui, app };
})();

document.addEventListener('DOMContentLoaded', ()=>OEQMS.app.init());
// expose for console debugging
window.OEQMS = OEQMS;
</script>
